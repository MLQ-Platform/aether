- S_open(i, t): Skewness of opening prices over 5-day window
- S_low_smooth(i, t): Exponentially smoothed skewness of daily low prices (α = 0.7)

import pandas as pd
import numpy as np

def factor(data: pd.DataFrame, params: dict = {}) -> pd.Series:
    window = params.get('window', 5)
    alpha = params.get('alpha', 0.7)
    
    s_open = data['OPEN'].rolling(window=window).skew()
    
    low_skew = data['LOW'].rolling(window=window).skew()
    s_low_smooth = low_skew.ewm(alpha=alpha, adjust=False).mean()
    
    result = s_open - s_low_smooth
    return result
